title GitHub Integration - Class Diagram

class GitHubClient {
    - baseUrl: String
    - accessToken: String
    + getRepositories(): List
    + getRepository(owner, repo): Repository
    + createRepository(name): Repository
}

class GitHubOAuth {
    + getAuthorizationUrl(): String
    + exchangeCode(code): Token
    + refreshToken(token): Token
}

class TokenManager {
    + saveToken(userId, token)
    + getToken(userId): Token
    + deleteToken(userId)
}

class Repository {
    - id: String
    - name: String
    - owner: String
    - url: String
}

class User {
    - id: String
    - email: String
    - username: String
}

class ForgeResolver {
    + connectGitHub()
    + getRepositories()
    + disconnectGitHub()
}

ForgeResolver --> GitHubOAuth
ForgeResolver --> GitHubClient
GitHubOAuth --> TokenManager
GitHubClient --> Repository
TokenManager --> User

@enduml
